NAME:
   aztfexport resource-group - Exporting a resource group and the nested resources resides within it.

USAGE:
   aztfexport resource-group [option] <resource group name>

OPTIONS:
   --name-pattern value, -p value                                             The pattern of the resource name. The pattern supports at most one index character, either '*' or '+' (exclusively): both expands to an incremental type-scoped index, '*' outputs no suffix for the first element, then 2, 3 and so on, where '+' output 1, 2, and so on. If none is specified, a '+' is implicitly appended at the end of the pattern. The pattern also supports a set of placeholders that are expanded per resource: {type} (the last Azure resource type segment, snake_cased, e.g. 'virtual_machines'), {rp} (the Azure resource provider namespace, snake_cased, e.g. 'microsoft_compute'), {name} (the last name segment of the Azure resource id, snake_cased), {root_scope} (the root scope of the resource, snake_cased, e.g. the resource group name). E.g. '{type}*' may expand to 'virtual_machines', 'virtual_machines2', ... (default: "res-") [%AZTFEXPORT_NAME_PATTERN%]
   --subscription-id value, -s value                                          The subscription id [%AZTFEXPORT_SUBSCRIPTION_ID%, %ARM_SUBSCRIPTION_ID%]
   --output-dir value, -o value                                               The output directory (will create the dir if it does not exist) (default: "D:\\Temp\\GitHub\\ITDM0007-IacTerrafAnsible\\ESERCIZI\\0004-HCL-Import") [%AZTFEXPORT_OUTPUT_DIR%]
   --overwrite, -f                                                            Proceed with non-empty output directory, which is likely to pollute the directory and cause errors (use with caution) (default: false) [%AZTFEXPORT_OVERWRITE%]
   --append                                                                   Imports to the existing state file if any and does not clean up the output directory (default: false) [%AZTFEXPORT_APPEND%]
   --dev-provider                                                             Use the local development provider, instead of the version pinned provider (default: false) [%AZTFEXPORT_DEV_PROVIDER%]
   --provider-version value                                                   The provider version to use for importing. Defaults to "5.11.0" for azurerm, 2.13.0 for azapi [%AZTFEXPORT_PROVIDER_VERSION%]
   --provider-name value                                                      The provider name to use for importing. Possible values are "azurerm" and "azapi". Defaults to "azurerm" (default: "azurerm") [%AZTFEXPORT_PROVIDER_NAME%]
   --backend-type value                                                       The Terraform backend used to store the state (default: local) [%AZTFEXPORT_BACKEND_TYPE%]
   --backend-config value [ --backend-config value ]                          The Terraform backend config [%AZTFEXPORT_BACKEND_CONFIG%]
   --config-mode value                                                        The trimming mode for the generated Terraform config. Can be one of "minimal" (most aggressive; trims zero values, schema defaults and Optional+Computed attributes), "lossless" (keeps Optional+Computed attributes so the config matches the live state) and "full" (keeps every property; may require manual edits to be valid) (default: "minimal") [%AZTFEXPORT_CONFIG_MODE%]
   --mask-sensitive                                                           Mask sensitive attributes in the Terraform configuration. This may require manual modifications to produce a valid config (default: false) [%AZTFEXPORT_MASK_SENSITIVE%]
   --parallelism value                                                        Limit the number of parallel operations, i.e., resource discovery, import (default: 10) [%AZTFEXPORT_PARALLELISM%]
   --non-interactive, -n                                                      Non-interactive mode (default: false) [%AZTFEXPORT_NON_INTERACTIVE%]
   --plain-ui                                                                 In non-interactive mode, print the progress information line by line, rather than the spinner UI. This can be used in OS that has no /dev/tty available (default: false) [%AZTFEXPORT_PLAIN_UI%]
   --continue, -k                                                             For non-interactive mode, continue on any import error (default: false) [%AZTFEXPORT_CONTINUE%]
   --generate-mapping-file, -g                                                Only generate the resource mapping file, but does NOT import any resource (default: false) [%AZTFEXPORT_GENERATE_MAPPING_FILE%]
   --hcl-only                                                                 Only generates HCL code (and mapping file), but not the files for resource management (e.g. the state file) (default: false) [%AZTFEXPORT_HCL_ONLY%]
   --module-path value                                                        The path of the module (e.g. "module1.module2") where the resources will be imported and config generated. Note that only modules whose "source" is local path is supported. Defaults to the root module. [%AZTFEXPORT_MODULE_PATH%]
   --generate-import-block                                                    Whether to generate the import.tf that contains the "import" blocks for the Terraform official plannable importing (default: false) [%AZTFEXPORT_GENERATE_IMPORT_BLOCK%]
   --log-path value                                                           The file path to store the log [%AZTFEXPORT_LOG_PATH%]
   --log-level value                                                          Log level, can be one of "ERROR", "WARN", "INFO", "DEBUG" and "TRACE" (default: "INFO") [%AZTFEXPORT_LOG_LEVEL%]
   --exclude-azure-resource value [ --exclude-azure-resource value ]          Exclude resource from being exported based on the Azure resource ID pattern (case-insensitive regexp) [%AZTFEXPORT_AZURE_RESOURCE%]
   --exclude-azure-resource-file value                                        Path to a file recording the excluded resources from being exported based on the Azure resource ID pattern (case-insensitive regexp), one per line [%AZTFEXPORT_AZURE_RESOURCE_FILE%]
   --exclude-terraform-resource value [ --exclude-terraform-resource value ]  Exclude resource from being exported based on the Terraform resource type [%AZTFEXPORT_TERRAFORM_RESOURCE%]
   --exclude-terraform-resource-file value                                    Path to a file recording the excluded resources from being exported based on the Terraform resource type, one per line [%AZTFEXPORT_TERRAFORM_RESOURCE_FILE%]
   --include-extension value [ --include-extension value ]                    Include extension resource types associated to the resources exported. Can be specified multiple times. Supported values: [role-assignments locks diagnostic-settings] [%AZTFEXPORT_INCLUDE_EXTENSION%]
   --include-managed-resource                                                 Whether to include internal resources managed by Azure in the exported configuration (default: false) [%AZTFEXPORT_INCLUDE_MANAGED_RESOURCE%]
   --env value                                                                The cloud environment, can be one of "public", "usgovernment" and "china" (default: "public") [%AZTFEXPORT_ENV%, %ARM_ENVIRONMENT%]
   --tenant-id value                                                          The Azure tenant id [%AZTFEXPORT_TENANT_ID%, %ARM_TENANT_ID%]

   --client-id value                                                          The Azure client id [%AZTFEXPORT_CLIENT_ID%, %ARM_CLIENT_ID%]
   --client-id-file-path value                                                The Azure client id file path [%AZTFEXPORT_CLIENT_ID_FILE_PATH%, %ARM_CLIENT_ID_FILE_PATH%]
   --client-certificate value                                                 A base64-encoded PKCS#12 bundle to be used as the client certificate for authentication [%AZTFEXPORT_CLIENT_CERTIFICATE%, %ARM_CLIENT_CERTIFICATE%]
   --client-certificate-path value                                            The path to the Client Certificate (PKCS#12) associated with the Service Principal which should be used [%AZTFEXPORT_CLIENT_CERTIFICATE_PATH%, %ARM_CLIENT_CERTIFICATE_PATH%]
   --client-certificate-password value                                        The password associated with the Client Certificate [%AZTFEXPORT_CLIENT_CERTIFICATE_PASSWORD%, %ARM_CLIENT_CERTIFICATE_PASSWORD%]
   --client-secret value                                                      The Azure client secret [%AZTFEXPORT_CLIENT_SECRET%, %ARM_CLIENT_SECRET%]
   --client-secret-file-path value                                            The Azure client secret file path [%AZTFEXPORT_CLIENT_SECRET_FILE_PATH%, %ARM_CLIENT_SECRET_FILE_PATH%]
   --oidc-request-token value                                                 The bearer token for the request to the OIDC provider [%AZTFEXPORT_OIDC_REQUEST_TOKEN%, %ARM_OIDC_REQUEST_TOKEN%, %ACTIONS_ID_TOKEN_REQUEST_TOKEN%]
   --oidc-request-url value                                                   The URL for the OIDC provider from which to request an ID token [%AZTFEXPORT_OIDC_REQUEST_URL%, %ARM_OIDC_REQUEST_URL%, %ACTIONS_ID_TOKEN_REQUEST_URL%]
   --oidc-token-file-path value                                               The path to a file containing an ID token when authenticating using OIDC [%AZTFEXPORT_OIDC_TOKEN_FILE_PATH%, %ARM_OIDC_TOKEN_FILE_PATH%]
   --oidc-token value                                                         The ID token when authenticating using OIDC [%AZTFEXPORT_OIDC_TOKEN%, %ARM_OIDC_TOKEN%]
   --use-managed-identity-cred                                                Explicitly use the managed identity that is provided by the Azure host to do authentication (default: false) [%AZTFEXPORT_USE_MANAGED_IDENTITY_CRED%, %ARM_USE_MSI%]
   --use-azure-cli-cred                                                       Explicitly use the Azure CLI to do authentication (default: true) [%AZTFEXPORT_USE_AZURE_CLI_CRED%, %ARM_USE_CLI%]
   --use-oidc-cred                                                            Explicitly use the OIDC to do authentication (default: false) [%AZTFEXPORT_USE_OIDC_CRED%, %ARM_USE_OIDC%]
   --help, -h                                                                 show help
